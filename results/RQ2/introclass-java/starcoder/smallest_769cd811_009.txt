    public int r, i;

        this.r = r;
        this.i = i;
    }

    return new Complex (a.r + b.r, a.i + b.i);

    return new Complex (a.r - b.r, a.i - b.i);

    return new Complex (a.r * b.r - a.i * b.i, a.i * b.r + a.r * b.i);
}

    return new Complex (a.r, -a.i);
}
public int magnitude (Complex a) {
    return (int) Math.sqrt (a.r * a.r + a.i * a.i);
}
