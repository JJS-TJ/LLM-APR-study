#include <stdio.h>
#include <stdlib.h>

int min(int x, int y){
	int z;
	if (x < y)
	z = x;
	else
	z = y;
	return z;
}

int main(){
	int x, y, z, t, a;
	printf("Please enter 4 numbers separated by spaces > ");
	scanf("%d %d %d %d", &x, &y, &z, &t);
	a = min(min(x, y), min(z, t));
