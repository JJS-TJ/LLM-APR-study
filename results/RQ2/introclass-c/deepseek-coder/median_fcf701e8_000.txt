#include <stdio.h>

int main () {

	int a, b, c, median;
     printf("Please enter three integers separated by spaces > ");
     scanf("%d%d%d", &a, &b, &c);
	
	/* The bug was that the three while loops checked if a,b,c were less than each other.
	Therefore, there was no way to find the median if all numbers were equal or a and b were equal.
	Additionally, there was no need to check whether b<c or a<c or c<a etc. in the while loops,
	because the median must be greater than or equal to one value and less than or equal to another,
	and we have already established this by our condition. 
	So, the code was only running the while loop once, and after finding the median,
	it was breaking out of the loop, and therefore only finding the median for the first input.
	
	The correct way to find the median is to check whether each number is greater than or equal to the other two,
	and also whether it is less than or equal to the other two. */
	
	if ((a >= b && a <= c) || (a >= c && a <= b)) {
		median = a;
	}
	else if ((b >= a && b <= c) || (b >= c && b <= a)) {
		median = b;
	}
	else {
		median = c;
	}

	printf("%d is the median\n", median);

return 0;
}
