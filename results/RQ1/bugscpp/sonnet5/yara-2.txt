    const uint8_t* data,
    size_t data_size,
    uint8_t* string,
    size_t string_length)
{

  const uint8_t* s2 = string;

  size_t i = 0;

  if (data_size < string_length * 2)
    return 0;

  while (i < string_length && s1[1] == 0 && yr_lowercase[*s1] == yr_lowercase[*s2])

  }

The bug was that the function wasn't checking the high byte of the wide character. Since `data` contains wide characters (2 bytes each) and `string` is single-byte, the comparison needs to verify that the high byte (`s1[1]`) is 0 to ensure we're comparing valid ASCII-range wide characters.
